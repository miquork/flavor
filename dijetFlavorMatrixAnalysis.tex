\begin{frame}{Pair-level IPF uses every stored count}
\begin{columns}[T,onlytextwidth]
\column{0.53\textwidth}
\begin{align*}
W_{\rm data} &= \arg\min_W D_{\rm KL}(W\Vert M),\\
\sum_F W(R,F) &= d_{\rm data}(R),\\
\sum_R W(R,F) &= \pi_{\rm MC}(F).
\end{align*}
\column{0.45\textwidth}
\begin{itemize}
\item $R$: 16 reconstructed tag/probe pairs
\item $F$: 16 true-flavor tag/probe pairs
\item counts recovered from profile bin entries
\item fixed MC truth fractions, as requested
\item marginal error below $10^{-10}$ in every pT range
\end{itemize}
\end{columns}
\vspace{0.4em}
\begin{block}{Requested single-jet outputs}
After the pair fit, tag and probe orientations are marginalized to obtain $\epsilon(t\mid f)$ and $P(f\mid t)$. Pair correlations are retained during inference.
\end{block}
\end{frame}

\begin{frame}{Dijets determine response differences, not an absolute scale}
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{align*}
R_{\rm HDM} &= \frac{m_0-m_n-m_u}
 {1-m_n/R_n-m_u/R_u},\\
q(R_{\rm HDM}) &= \frac{R_{\rm HDM}+1}{3-R_{\rm HDM}},\\
q_{\rm data}(R,F) &= q_{\rm MC}(R,F)
\frac{k_{f_{\rm probe}}}{k_{f_{\rm tag}}}.
\end{align*}
\column{0.44\textwidth}
\begin{block}{Fit choices}
\begin{itemize}
\item $R_n=1.00$, $R_u=0.92$
\item merged $m_0,m_n,m_u$ component means
\item all MC reco/true response cells retained
\item one $k_f$ per true flavor across reco cells
\item $k_{uds}=1$ fixes the unobservable common mode
\item nonlinear least squares, both orientations
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Recommended next step: add dijet as a relative-response constraint}
\begin{enumerate}
\item Keep \texttt{ab} as the central dijet response coordinate; use \texttt{ad/tc/pf} for migration and method systematics.
\item Add calibrated tag/mistag priors and float truth fractions before interpreting the IPF efficiency shifts as SFs.
\item Build one likelihood with Z+jet/$\gamma$+jet absolute reference information and dijet high-pT relative-response information.
\item Preserve independent 2024/2025/2026 detector fits under common smooth extrapolation forms.
\item Add ud/s separation only when W$\to cs$ inputs are available; keep the present $uds$ result as the four-flavor baseline.
\end{enumerate}
